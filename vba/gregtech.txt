option VBASupport 1

function POWER_LEVEL(level as string, optional withULV) as integer
	select case LCase(Trim(level))
		case "ulv" : POWER_LEVEL = 0
		case "lv"  : POWER_LEVEL = 1
		case "mv"  : POWER_LEVEL = 2
		case "hv"  : POWER_LEVEL = 3
		case "ev"  : POWER_LEVEL = 4
		case "iv"  : POWER_LEVEL = 5
		case "luv" : POWER_LEVEL = 6
		case "zpm" : POWER_LEVEL = 7
		case "uv"  : POWER_LEVEL = 8
		case "uhv" : POWER_LEVEL = 9
		case "uev" : POWER_LEVEL = 10
		case "uiv" : POWER_LEVEL = 11
		case "uxv" : POWER_LEVEL = 12
		case "opv" : POWER_LEVEL = 13
		case else  : POWER_LEVEL = -10
	end select
	if isMissing(withULV) and POWER_LEVEL = 0 then
		POWER_LEVEL = 1
	end if
end function

function OVERCLOCKS(level as string, minLevel as string, optional withULV) as integer
	OVERCLOCKS = POWER_LEVEL(level, withULV) - POWER_LEVEL(minLevel, withULV)
end function

function TIME_NORMAL_DIFF(level as string, minLevel as string, optional withULV) as double
	TIME_NORMAL_DIFF = 2 ^ OVERCLOCKS(level, minLevel, withULV)
end function

function TIME_PERFECT_DIFF(level as string, minLevel as string, optional withULV) as double
	TIME_PERFECT_DIFF = 4 ^ OVERCLOCKS(level, minLevel, withULV)
end function

function NORMAL_CALCULATED(time as double, level as string, minLevel as string, optional withULV) as double
on local error resume next
	NORMAL_CALCULATED = time / TIME_NORMAL_DIFF(level, minLevel, withULV)
end function

function PERFECT_CALCULATED(time as double, level as string, minLevel as string, optional withULV) as double
on local error resume next
	PERFECT_CALCULATED = time / TIME_PERFECT_DIFF(level, minLevel, withULV)
end function

function NORMAL_TIME(time as double, level as string, minLevel as string, optional withULV) as double
	NORMAL_TIME = MINECRAFT_TIME(NORMAL_CALCULATED(time, level, minLevel, withULV))
end function

function PERFECT_TIME(time as double, level as string, minLevel as string, optional withULV) as double
	PERFECT_TIME = MINECRAFT_TIME(PERFECT_CALCULATED(time, level, minLevel, withULV))
end function

function PYROLYSE_TIME(time as double, level as string, minLevel as string, coil as string) as double
on local error resume next
	PYROLYSE_TIME = MINECRAFT_TIME(NORMAL_CALCULATED(time, level, minLevel) / PYROLYSE_SPEED(coil))
end function

function REACTOR_TIME(time as double, level as string, minLevel as string, coil as string, optional para) as double
on local error resume next
	if isMissing(para) then
		REACTOR_TIME = MINECRAFT_TIME(NORMAL_CALCULATED(time, level, minLevel) / REACTOR_SPEED(coil))
	else
		REACTOR_TIME = MINECRAFT_TIME(NORMAL_CALCULATED(time, level, minLevel) / REACTOR_SPEED(coil) * INT(SQR(para)))
	end if
end function

function MULTI_TIME(time as double, level as string, minLevel as string, coil as string) as double
on local error resume next
	MULTI_TIME = MINECRAFT_TIME(NORMAL_CALCULATED(time, level, minLevel) / MULTI_SPEED(coil))
end function

function CAPABILITY(level as string, minLevel as string, amps as double) as long
on local error resume next
	CAPABILITY = INT(TIME_PERFECT_DIFF(level, minLevel, true) / amps)
end function

function PARALLEL_COUNT(level as string, minLevel as string, amps as double, para as long) as long
	PARALLEL_COUNT = INT(MINIMUM(CAPABILITY(level, minLevel, amps), para))
end function

function PARALLEL_TIME(time as double, level as string, minLevel as string, para as long) as double
on local error resume next
	PARALLEL_TIME = MINECRAFT_TIME(NORMAL_CALCULATED(time, level, minLevel, withULV) * INT(SQR(para)))
end function

function ENERGY_LEVEL(level as integer) as string
	select case level
		case 0    : ENERGY_LEVEL = "ulv"
		case 1    : ENERGY_LEVEL = "lv"
		case 2    : ENERGY_LEVEL = "mv"
		case 3    : ENERGY_LEVEL = "hv"
		case 4    : ENERGY_LEVEL = "ev"
		case 5    : ENERGY_LEVEL = "iv"
		case 6    : ENERGY_LEVEL = "luv"
		case 7    : ENERGY_LEVEL = "zpm"
		case 8    : ENERGY_LEVEL = "uv"
		case 9    : ENERGY_LEVEL = "uhv"
		case 10   : ENERGY_LEVEL = "uev"
		case 11   : ENERGY_LEVEL = "uiv"
		case 12   : ENERGY_LEVEL = "uxv"
		case 13   : ENERGY_LEVEL = "opv"
		case else : ENERGY_LEVEL = "unknown"
	end select
end function

function MINECRAFT_TIME(time as double) as double 
	dim ticks as long
	ticks = INT(time * 20)
	if ticks <= 1 then
		ticks = 1
	end if
	MINECRAFT_TIME = ticks / 20.0
end function

function UNIT_COUNT(count as double, unit as double) as double
	dim res as long
	res = INT(count / unit)
	if res < 1 then
		res = 1
	end if
	UNIT_COUNT = res * unit
end function

function ENERGY_BASE(energy as double) as string
	ENERGY_BASE = ENERGY_LEVEL(CEILING(LOG(energy, 2)))
end function

function MINIMUM(x, y as variant) as variant
   MINIMUM = IIF(x < y, x, y)
end function
